### Domainwright: triggers shared by the interaction, the menu and the window.

### Root: the character whose limit would be edited.
### Alive is the only hard requirement; everything else is access policy, checked by the
### interaction because only it knows who the actor is.
dlc_target_is_editable_trigger = {
	is_alive = yes
}

### True when the override is non-zero and therefore worth showing or resetting.
dlc_has_override_trigger = {
	exists = var:dlc_delta
	NOT = { var:dlc_delta = 0 }
}

### Root: the acting character (the interaction's actor). scope:recipient is the target.
### Implements the dlc_access game rule.
###
### The settings are spelled out as a flat OR rather than an if/else_if chain: `else_if` is
### not a valid key inside a scripted trigger. 1.20.1.1 reports
### "Unknown trigger: else_if" and then discards the whole trigger, which made the
### interaction permanently invalid.
###
### Every branch tests the target alone - never the actor. `root` is not the actor in the
### pass that builds the interaction's failure tooltips, and comparing against it there
### produced "Event target link 'root' returned an unset scope" 99,729 times in one
### session. In single-player the three settings reduce to a property of the target:
### anyone, the player, or the player plus their realm.
###
### "Is the player" is spelled `is_ai = no`. `is_player` is not a trigger in 1.20.1.1: at
### load the engine reports "Unknown trigger: is_player" and substitutes false, which left
### the player_only and player_and_realm settings matching nobody while the default rule
### kept working and hid the fault.
dlc_actor_may_edit_recipient_trigger = {
	scope:recipient = { is_alive = yes }

	OR = {
		has_game_rule = dlc_access_anyone

		AND = {
			has_game_rule = dlc_access_player_only
			scope:recipient = { is_ai = no }
		}

		AND = {
			has_game_rule = dlc_access_player_and_realm
			scope:recipient = {
				OR = {
					is_ai = no
					AND = {
						exists = top_liege
						top_liege = { is_ai = no }
					}
				}
			}
		}
	}
}

### True when the character's government can clamp the effective limit, making the upper
### rungs inert. `government_uses_domain_limit` is a government flag declared in each
### government's `flags = { }` block (common/governments/00_government_types.txt:34 and
### elsewhere), and its vanilla comment defines it as true for governments that do NOT use
### `domain_limit_max` or that hold no land. So the cap risk is the flag's *absence*;
### landless rulers are already excluded because the flag is true for them.
###
### Capped in 1.20.1.1: nomadic authority laws set 1-3, Ritsuryo and Soryo sit at 1-2 and are
### raised only by the Japanese manor, Mandala and Wanua cap at 1 and are raised by their own
### buildings, and two governments cap at 2 outright (00_government_types.txt:297 and :1056).
### Warn rather than clamp: the ladder is not wrong there, it is outranked.
dlc_government_may_cap_limit_trigger = {
	NOT = { government_has_flag = government_uses_domain_limit }
}
