### Domainwright: scripted effects for the delta.
###
### Invariants:
###   1. var:dlc_delta is the only persisted variable (absent means 0).
###   2. The set of ladder modifiers on a character is always exactly the bits of dlc_delta.
###   3. Every effect here is safe to run repeatedly; none of them stack.
###   4. No `limit` in this file reads a variable, and no local variable is used at all.
###
### Invariant 4 is the one that keeps R12 clean. While an event option or an interaction is
### described, the engine dry-runs the effects behind it: it evaluates every `limit`, including
### those inside effects reached through `trigger_event`, `hidden_effect` and `show_as_tooltip`,
### and it executes nothing. A limit that read a variable therefore hit an unset one on every
### refresh - 47,817 lines in one session from a local, then 249 more once the ladder was
### reached through the interaction's on_accept. The bit tests come from
### common/script_values/dlc_domain_limit_values.txt, which reads var:dlc_delta only behind an
### `exists` check.

### Strips every rung. Guarded so a character that never had an override logs nothing.
dlc_clear_ladder_effect = {
	if = {
		limit = { has_character_modifier = dlc_domain_limit_p1 }
		remove_character_modifier = dlc_domain_limit_p1
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_p2 }
		remove_character_modifier = dlc_domain_limit_p2
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_p4 }
		remove_character_modifier = dlc_domain_limit_p4
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_p8 }
		remove_character_modifier = dlc_domain_limit_p8
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_p16 }
		remove_character_modifier = dlc_domain_limit_p16
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_p32 }
		remove_character_modifier = dlc_domain_limit_p32
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_p64 }
		remove_character_modifier = dlc_domain_limit_p64
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_n1 }
		remove_character_modifier = dlc_domain_limit_n1
	}
	if = {
		limit = { has_character_modifier = dlc_domain_limit_n2 }

	if = {
		limit = { has_character_modifier = dlc_domain_limit_n32 }
		remove_character_modifier = dlc_domain_limit_n32
	}
}

### Rebuilds the ladder from var:dlc_delta.
###
### One block per rung, positive and negative side by side. The greedy subtraction the earlier
### version used needed a running remainder in a local variable, and the remainder's limits
### were what the describe pass could not evaluate. Reading each bit independently needs no
### running state: the set of rungs is a function of the delta alone, which is also the
### invariant this file promises.
dlc_apply_delta_effect = {
	dlc_clear_ladder_effect = yes

	if = {
		limit = {
			dlc_delta_bit64_value >= 64
			NOT = { dlc_delta_value < 0 }
		}
		add_character_modifier = dlc_domain_limit_p64
	}
	if = {
		limit = {
			dlc_delta_bit32_value >= 32
			NOT = { dlc_delta_value < 0 }
		}
		add_character_modifier = dlc_domain_limit_p32
	}
	if = {
		limit = {
			dlc_delta_bit32_value >= 32
			dlc_delta_value < 0
		}
		add_character_modifier = dlc_domain_limit_n32
	}
	if = {
		limit = {
			dlc_delta_bit16_value >= 16
			NOT = { dlc_delta_value < 0 }
		}
		add_character_modifier = dlc_domain_limit_p16
	}
	if = {
		limit = {
			dlc_delta_bit16_value >= 16
			dlc_delta_value < 0
		}
		add_character_modifier = dlc_domain_limit_n16
	}
	if = {
		limit = {
			dlc_delta_bit8_value >= 8
			NOT = { dlc_delta_value < 0 }
		}
		add_character_modifier = dlc_domain_limit_p8
	}
	if = {
		limit = {
			dlc_delta_bit8_value >= 8
			dlc_delta_value < 0
		}
		add_character_modifier = dlc_domain_limit_n8
	}
	if = {
		limit = {
			dlc_delta_bit4_value >= 4
			NOT = { dlc_delta_value < 0 }
		}
		add_character_modifier = dlc_domain_limit_p4
	}
	if = {
		limit = {
			dlc_delta_bit4_value >= 4
			dlc_delta_value < 0
		}
		add_character_modifier = dlc_domain_limit_n4
	}
	if = {
		limit = {
			dlc_delta_bit2_value >= 2
			NOT = { dlc_delta_value < 0 }
		}
		add_character_modifier = dlc_domain_limit_p2
	}
	if = {
		limit = {
			dlc_delta_bit2_value >= 2
			dlc_delta_value < 0
		}
		add_character_modifier = dlc_domain_limit_n2
	}
	if = {
		limit = {
			dlc_delta_bit1_value >= 1
			NOT = { dlc_delta_value < 0 }
		}
		add_character_modifier = dlc_domain_limit_p1
	}
	if = {
		limit = {
			dlc_delta_bit1_value >= 1
			dlc_delta_value < 0
		}
		add_character_modifier = dlc_domain_limit_n1
	}
}

### Clamps, stores and applies. $VALUE$ may be a number or a script value block.
###
### The clamps read script values for the same reason the ladder does; the delta is written
### first, so they judge the value that is about to be applied.
dlc_set_delta_effect = {
	set_variable = {
		name = dlc_delta
		value = $VALUE$
	}

	if = {
		limit = { dlc_delta_value > 127 }
		set_variable = {
			name = dlc_delta
			value = 127
		}
	}
	else_if = {
		limit = { dlc_delta_value < -63 }
		set_variable = {
			name = dlc_delta
			value = -63
		}
	}
	else_if = {
		limit = { dlc_delta_value = 0 }
		remove_variable = dlc_delta
	}

	dlc_apply_delta_effect = yes
}

### Adds a step to the current delta. $VALUE$ is the step, e.g. 1 or -5.
###
### The arithmetic is done on the persisted variable rather than a local one: a parameter
### substitution inserts the argument's inner text only - braces and all - so passing a
### script-value block as `VALUE = { value = ... add = ... }` lands in the callee as the
### malformed assignment `value = value = var:dlc_delta add = 1`, and 1.20.1.1 reports
### "Compiling source for dlc_set_delta_effect failed for unknown arguments: value, add".
dlc_change_delta_effect = {
	if = {
		limit = { exists = var:dlc_delta }
		change_variable = {
			name = dlc_delta
			add = $VALUE$
		}
	}
	else = {
		set_variable = {
			name = dlc_delta
			value = $VALUE$
		}
	}

	dlc_set_delta_effect = { VALUE = var:dlc_delta }
}

### Clears one character's override.
dlc_reset_effect = {
	dlc_clear_ladder_effect = yes
	remove_variable = dlc_delta
}

### Clears every override in the world. Expensive: run from the menu, never on a pulse.
### Characters without var:dlc_delta are skipped, which is nearly all of them.
dlc_reset_all_effect = {
	every_living_character = {
		limit = { exists = var:dlc_delta }
		dlc_reset_effect = yes
	}
}

### Succession inheritance, called from common/on_action/dlc_on_actions.txt.
### Only does anything when the game rule asks for it.
dlc_inherit_on_succession_effect = {
	if = {
		limit = { has_game_rule = dlc_inheritance_heir_inherits_delta }
		if = {
			limit = { exists = scope:dlc_predecessor }
			if = {
				limit = { scope:dlc_predecessor = { exists = var:dlc_delta } }
				dlc_set_delta_effect = {
					VALUE = scope:dlc_predecessor.var:dlc_delta
				}
			}
		}
	}
}
